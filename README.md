        }

        // تابع بازگشت به محتوای اصلی
        window.showMainContent = function() {
            // بازسازی ساختار اصلی برای نمایش مجدد گرید یا نتایج جستجو
            contentArea.innerHTML = `
                <section id="postsContainer" class="posts-grid" style="display: ${searchInput.value.trim() ? 'none' : 'grid'};">
                </section>
                <section id="searchResults" class="search-results-section" style="display:${searchInput.value.trim() ? 'block' : 'none'};">
                    <h2>نتایج جستجو:</h2>
                    <div id="searchResultsContainer" class="posts-grid">
                    </div>
                </section>
            `;

            // گرفتن مجدد المان‌ها بعد از بازسازی
            const newPostsContainer = document.getElementById('postsContainer');
            const newSearchResultsContainer = document.getElementById('searchResultsContainer');
            
            if (searchInput.value.trim()) {
                // اگر جستجو فعال بود، نتایج را مجدداً رندر کن
                searchPosts(); 
            } else {
                // در غیر این صورت، همه پست‌ها را رندر کن
                renderPosts(allPosts, newPostsContainer);
            }

            window.scrollTo(0, 0);
        }

        // تابع جستجو (نزدیک‌ترین عنوان)
        window.searchPosts = function() {
            const query = searchInput.value.trim().toLowerCase();
            
            document.getElementById('postsContainer').style.display = 'none';
            searchResultsSection.style.display = 'block';

            if (!query) {
                document.getElementById('postsContainer').style.display = 'grid';
                searchResultsSection.style.display = 'none';
                renderPosts(allPosts, postsContainer);
                return;
            }

            const scoredPosts = allPosts.map(post => {
                let score = 0;
                const postTitle = post.title.toLowerCase();
                const postTags = post.tags.join(' ').toLowerCase();
                // تفکیک کوئری به کلمات برای امتیازدهی دقیق‌تر
                const queryWords = query.split(/\s+/).filter(w => w.length > 1);

                queryWords.forEach(word => {
                    if (postTitle.includes(word)) {
                        score += 3;
                    }
                    if (postTags.includes(word)) {
                        score += 1;
                    }
                });

                return { ...post, score };
            });

            const filteredAndSorted = scoredPosts
                .filter(p => p.score > 0)
                .sort((a, b) => b.score - a.score);

            renderPosts(filteredAndSorted, searchResultsContainer);
            
            // اسکرول به بخش نتایج جستجو
            document.getElementById('searchResults').scrollIntoView({ behavior: 'smooth' });
        }

        // بارگذاری اولیه
        document.addEventListener('DOMContentLoaded', () => {
            document.getElementById('postsContainer').style.display = 'grid';
            searchResultsSection.style.display = 'none';
            renderPosts(allPosts, postsContainer);
        });
    </script>
